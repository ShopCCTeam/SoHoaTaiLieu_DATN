### Bước 4 — Implement `security.py`

```python
"""Auth security primitives — argon2id (pwdlib) + JWT (PyJWT)."""
from __future__ import annotations

import hashlib
import secrets
import time
from typing import Any

import jwt
from pwdlib import PasswordHash

from app.core.config import get_settings

# argon2id recommended params (OWASP 2024).
_pwd = PasswordHash.recommended()

# Valid argon2id dummy hash for timing defense.
# Hash of "dummy-password-never-used" — never used in practice.
DUMMY_PASSWORD_HASH: str = _pwd.hash("dummy-password-never-used")


def hash_password(plain: str) -> str:
    """Hash password với argon2id. Plain text KHÔNG log."""
    return _pwd.hash(plain)


def verify_password(plain: str, hashed: str) -> bool:
    """Verify plain password. False nếu hash malformed."""
    if not hashed:
        return False
    try:
        return _pwd.verify(plain, hashed)
    except (ValueError, TypeError):
        return False


# ---- Refresh token (opaque random, SHA-256 hash lưu DB) ----

def generate_refresh_token() -> tuple[str, str]:
    """Sinh opaque refresh token. Return (raw_token, sha256_hex_hash)."""
    raw = secrets.token_urlsafe(48)  # 384 bits
    return raw, hashlib.sha256(raw.encode("utf-8")).hexdigest()


def hash_refresh_token(raw: str) -> str:
    """SHA-256 hex của raw refresh token."""
    return hashlib.sha256(raw.encode("utf-8")).hexdigest()


# ---- Access token (JWT HS256) ----

def create_access_token(*, subject: str, role: str, ttl_seconds: int | None = None) -> str:
    settings = get_settings()
    ttl = ttl_seconds if ttl_seconds is not None else settings.jwt_access_token_ttl_seconds
    now = int(time.time())
    payload: dict[str, Any] = {
        "sub": subject,
        "role": role,
        "iat": now,
        "exp": now + ttl,
        "iss": settings.app_name,
        "type": "access",
    }
    return jwt.encode(payload, settings.jwt_secret.get_secret_value(), algorithm=settings.jwt_algorithm)


def decode_access_token(token: str) -> dict[str, Any]:
    settings = get_settings()
    return jwt.decode(
        token,
        settings.jwt_secret.get_secret_value(),
        algorithms=[settings.jwt_algorithm],
        options={"require": ["exp", "sub", "role"]},
    )
```

## Commit 4: feat(auth): refresh rotation service (transaction ownership rõ ràng)

- Service layer **không commit**. Trả raw objects/results.
- Router/service gọi `session.commit()` một lần.
- User loaded rõ ràng: `session.get(User, user_id)` trong transaction.

```python
"""Refresh token rotation + reuse detection.

Transaction ownership:
  - Helper/service KHÔNG tự commit.
  - Router/service gọi session.commit() một lần sau khi gọi helper.
  - Reuse detection commit revocation TRƯỚC KHI raise AuthRefreshError.

User loading:
  - Service trả RefreshResult (raw_token, family_id, user_id).
  - Router/service load user rõ ràng bằng session.get(User, user_id).
"""
from __future__ import annotations

import uuid
from dataclasses import dataclass
from datetime import datetime, timedelta, timezone

from sqlalchemy import select, update
from sqlalchemy.ext.asyncio import AsyncSession

from app.models.refresh_session import RefreshSession
from app.modules.auth.security import generate_refresh_token, hash_refresh_token


class AuthRefreshError(Exception):
    """Refresh rotation failed.

    Attributes:
        detail: Human-readable message.
        code: AUTH_REFRESH_* error code.
    """

    def __init__(self, detail: str, code: str) -> None:
        super().__init__(detail)
        self.detail = detail
        self.code = code


@dataclass
class RefreshResult:
    """Kết quả rotation — chưa commit."""
    raw_token: str
    family_id: uuid.UUID
    user_id: str
    expires_at: datetime


async def create_session_for_login(
    session: AsyncSession,
    *,
    user_id: str,
    raw_token: str,
    token_hash: str,
    ttl_seconds: int,
    user_agent: str | None = None,
    ip_address: str | None = None,
) -> uuid.UUID:
    """Tạo family mới + session đầu tiên. KHÔNG commit.

    Return: family_id (UUID)
    """
    family_id = uuid.uuid4()
    expires_at = datetime.now(timezone.utc) + timedelta(seconds=ttl_seconds)
    obj = RefreshSession(
        id=uuid.uuid4(),
        user_id=user_id,
        family_id=family_id,
        token_hash=token_hash,
        user_agent=user_agent,
        ip_address=ip_address,
        expires_at=expires_at,
    )
    session.add(obj)
    await session.flush()
    return family_id


async def rotate_session(
    session: AsyncSession,
    *,
    raw_token: str,
    ttl_seconds: int,
    user_agent: str | None = None,
    ip_address: str | None = None,
) -> RefreshResult:
    """Atomic rotate: lookup → validate → revoke old → create new.

    KHÔNG commit. Gọi session.commit() ở router/service.
    Raises AuthRefreshError (sau khi commit revocation nếu reuse).
    """
    submitted_hash = hash_refresh_token(raw_token)
    now = datetime.now(timezone.utc)

    # 1. Tìm session bằng token_hash (FOR UPDATE lock)
    stmt = (
        select(RefreshSession)
        .where(RefreshSession.token_hash == submitted_hash)
        .with_for_update()
    )
    result = await session.execute(stmt)
    found = result.scalar_one_or_none()

    if found is None:
        raise AuthRefreshError(
            detail="Refresh token không hợp lệ.",
            code="AUTH_REFRESH_INVALID",
        )

    family_id = found.family_id
    user_id = found.user_id

    # 2. EXPIRED check
    if found.expires_at <= now:
        raise AuthRefreshError(
            detail="Refresh token đã hết hạn.",
            code="AUTH_REFRESH_EXPIRED",
        )

    # 3. REUSE DETECTION — token đã revoked
    if found.revoked_at is not None:
        # Commit family revocation TRƯỚC KHI raise
        await _revoke_family_no_flush(session, family_id, now)
        await session.commit()
        raise AuthRefreshError(
            detail="Phát hiện reuse refresh token — toàn bộ session bị thu hồi.",
            code="AUTH_REFRESH_REUSE_DETECTED",
        )

    # 4. OK — rotate: create new session, atomic update old
    new_raw, new_hash = generate_refresh_token()
    new_session = RefreshSession(
        id=uuid.uuid4(),
        user_id=user_id,
        family_id=family_id,
        token_hash=new_hash,
        user_agent=user_agent,
        ip_address=ip_address,
        issued_at=now,
        expires_at=now + timedelta(seconds=ttl_seconds),
    )
    session.add(new_session)

    # Atomic UPDATE: chỉ revoke row nếu chưa revoked (double-check dưới lock)
    update_result = await session.execute(
        update(RefreshSession)
        .where(
            RefreshSession.id == found.id,
            RefreshSession.revoked_at.is_(None),
        )
        .values(revoked_at=now, replaced_by_id=new_session.id)
    )
    if update_result.rowcount == 0:
        # Concurrent request đã revoke row → reuse detection
        await session.rollback()
        await _revoke_family_no_flush(session, family_id, now)
        await session.commit()
        raise AuthRefreshError(
            detail="Phát hiện reuse refresh token — toàn bộ session bị thu hồi.",
            code="AUTH_REFRESH_REUSE_DETECTED",
        )

    # KHÔNG commit — router/service gọi commit
    return RefreshResult(
        raw_token=new_raw,
        family_id=family_id,
        user_id=user_id,
        expires_at=new_session.expires_at,
    )


async def lookup_session_by_token(session: AsyncSession, raw_token: str) -> RefreshSession | None:
    """Lookup session bằng token_hash. KHÔNG commit."""
    submitted_hash = hash_refresh_token(raw_token)
    stmt = select(RefreshSession).where(RefreshSession.token_hash == submitted_hash)
    result = await session.execute(stmt)
    return result.scalar_one_or_none()


async def revoke_family(
    session: AsyncSession,
    *,
    family_id: uuid.UUID,
) -> None:
    """Thu hồi toàn bộ family. KHÔNG commit.

    Router/service gọi session.commit() sau khi revoke.
    """
    now = datetime.now(timezone.utc)
    await _revoke_family_no_flush(session, family_id, now)


async def _revoke_family_no_flush(session: AsyncSession, family_id: uuid.UUID, now: datetime) -> None:
    """Internal: revoke all active sessions in family. Does NOT flush/commit."""
    await session.execute(
        update(RefreshSession)
        .where(
            RefreshSession.family_id == family_id,
            RefreshSession.revoked_at.is_(None),
        )
        .values(revoked_at=now)
    )
```

```python
    *,

    reason: str | None = None,
    ip: str | None = None,

        reason=reason,
        ip=ip,
        user_agent=user_agent,
    )
    if level == "error":

    elif level == "warning":
        log.warning(event)
    else:
        log.info(event)


def require_origin(request: Request) -> None:
    """Verify Origin header khớp APP_CORS_ORIGINS.

    Raise 403 nếu Origin missing hoặc không khớp ALLOWED_ORIGINS.
    Bỏ qua nếu APP_ENV = development/test (localhost luôn được).
    """
    settings = get_settings()
    if settings.app_env in {"development", "test"}:
        return  # Development: bỏ qua Origin check

    origin = request.headers.get("origin") or request.headers.get("Origin")
    if not origin:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail={
                "code": "AUTH_CSRF_MISSING_ORIGIN",
                "detail": "Origin header missing.",
            },
        )

    try:
        from urllib.parse import urlparse
        origin_host = urlparse(origin).netloc
    except Exception:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail={
                "code": "AUTH_CSRF_MISSING_ORIGIN",
                "detail": "Invalid Origin header.",
            },
        )

    allowed_hosts = {urlparse(o).netloc for o in settings.app_cors_origins}
    if origin_host not in allowed_hosts:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail={
                "code": "AUTH_CSRF_MISSING_ORIGIN",
                "detail": f"Origin {origin} không được phép.",
```

```python
from pydantic import model_validator
from urllib.parse import urlparse

@model_validator(mode="after")
def _validate_production_and_staging(self) -> "Settings":
    if self.app_env not in {"production", "staging"}:
        return self

    # JWT secret
    jwt_value = self.jwt_secret.get_secret_value()
    if jwt_value.startswith("dev-only") or len(jwt_value) < 32:
        raise ValueError(
            "JWT_SECRET phải ≥ 32 bytes và KHÔNG được là default dev-only "
            f"khi APP_ENV={self.app_env}."
        )

    # Postgres password
    if self.postgres_password.get_secret_value() == "change-me":
        raise ValueError(
            f"POSTGRES_PASSWORD không được là 'change-me' khi APP_ENV={self.app_env}."
        )

    # MinIO secret
    if self.minio_secret_key.get_secret_value() == "minioadmin":
        raise ValueError(
            f"MINIO_SECRET_KEY không được là 'minioadmin' khi APP_ENV={self.app_env}."
        )

    # Cookie secure
    if not self.refresh_cookie_secure:
        raise ValueError(
            f"REFRESH_COOKIE_SECURE phải là true khi APP_ENV={self.app_env} (HTTPS bắt buộc)."
        )

    # CORS origins — không localhost/wildcard ở production/staging
    for origin in self.app_cors_origins:
        if "localhost" in origin or origin == "*":
            raise ValueError(
                f"APP_CORS_ORIGINS không được chứa localhost/wildcard khi APP_ENV={self.app_env}: "
                f"{origin}"
            )
        # Validate URL format
        try:
            parsed = urlparse(origin)
            if not parsed.scheme or not parsed.netloc:
                raise ValueError(f"Origin phải là URL hợp lệ: {origin}")
        except Exception as exc:
            raise ValueError(f"APP_CORS_ORIGINS URL không hợp lệ: {origin} — {exc}")

    return self
```
