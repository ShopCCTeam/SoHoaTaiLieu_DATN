```python
"""Auth security primitives — argon2id (pwdlib) + JWT (PyJWT).

Tất cả password hash là argon2id với params m=64MB, t=3, p=4.
"""
from __future__ import annotations

import hashlib
import secrets
import time
from typing import Any

import jwt
from pwdlib import PasswordHash

from app.core.config import get_settings

# argon2id recommended params (OWASP 2024).
_pwd = PasswordHash.recommended()  # m=64MB, t=3, p=4


def hash_password(plain: str) -> str:
    """Hash password với argon2id. Plain text KHÔNG log."""
    return _pwd.hash(plain)


def verify_password(plain: str, hashed: str) -> bool:
    """Verify plain password. Constant-time. False nếu hash malformed."""
    try:
        return _pwd.verify(plain, hashed)
    except (ValueError, TypeError):
        return False


# ---- Refresh token (opaque random, SHA-256 hash lưu DB) ----

def generate_refresh_token() -> tuple[str, str]:
    """Sinh opaque refresh token. Return (raw_token, sha256_hex_hash).

    raw_token gửi qua HttpOnly cookie; hash lưu DB. Không lưu raw.
    """
    raw = secrets.token_urlsafe(48)  # 384 bits entropy
    return raw, hashlib.sha256(raw.encode("utf-8")).hexdigest()


def hash_refresh_token(raw: str) -> str:
    """SHA-256 hex của raw refresh token."""
    return hashlib.sha256(raw.encode("utf-8")).hexdigest()


# ---- Access token (JWT HS256) ----

def create_access_token(*, subject: str, role: str, ttl_seconds: int | None = None) -> str:
    """Tạo HS256 access token."""
    settings = get_settings()
    ttl = ttl_seconds if ttl_seconds is not None else settings.jwt_access_token_ttl_seconds
    now = int(time.time())
    payload: dict[str, Any] = {
        "sub": subject,
        "role": role,
        "iat": now,
        "exp": now + ttl,
        "iss": settings.app_name,
        "type": "access",
    }
    return jwt.encode(
        payload,
        settings.jwt_secret.get_secret_value(),
        algorithm=settings.jwt_algorithm,
    )


def decode_access_token(token: str) -> dict[str, Any]:
    """Decode + verify access token. Raises `jwt.PyJWTError` nếu invalid/expired."""
    settings = get_settings()
    return jwt.decode(
        token,
        settings.jwt_secret.get_secret_value(),
        algorithms=[settings.jwt_algorithm],
        options={"require": ["exp", "sub", "role"]},
    )
```

```python
"""RefreshSession ORM — lưu refresh token opaque hash + family revocation.

Mỗi lần login → tạo 1 family mới + 1 session. Mỗi rotate → set revoked_at
cho session cũ, tạo session mới cùng family_id, set replaced_by_id.

Reuse detection: nếu session đã revoked_at mà vẫn bị submit → thu hồi
TOÀN BỘ family (set revoked_at cho mọi session thuộc family đó).
"""
from __future__ import annotations

import uuid
from datetime import datetime

from sqlalchemy import DateTime, ForeignKey, String, func
from sqlalchemy.dialects.postgresql import INET, UUID
from sqlalchemy.orm import Mapped, mapped_column

from app.db.base import Base


class RefreshSession(Base):
    __tablename__ = "refresh_sessions"

    id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True), primary_key=True, server_default=func.gen_random_uuid()
    )
    user_id: Mapped[str] = mapped_column(
        String(36), ForeignKey("users.id", ondelete="CASCADE"), nullable=False, index=True
    )
    family_id: Mapped[uuid.UUID] = mapped_column(UUID(as_uuid=True), nullable=False, index=True)
    token_hash: Mapped[str] = mapped_column(String(64), nullable=False, unique=True)
    user_agent: Mapped[str | None] = mapped_column(String(512), nullable=True)
    ip_address: Mapped[str | None] = mapped_column(INET, nullable=True)
    issued_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True), nullable=False, server_default=func.now()
    )
    expires_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True), nullable=False, index=True
    )
    revoked_at: Mapped[datetime | None] = mapped_column(
        DateTime(timezone=True), nullable=True, index=True
    )
    replaced_by_id: Mapped[uuid.UUID | None] = mapped_column(
        UUID(as_uuid=True),
        ForeignKey("refresh_sessions.id", ondelete="SET NULL"),
        nullable=True,
    )
    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True), nullable=False, server_default=func.now()
    )

    def __repr__(self) -> str:  # pragma: no cover
        return f"<RefreshSession id={self.id} family={self.family_id} user={self.user_id}>"
```

**Step 3 — Implement service**

```python
"""Refresh token rotation + reuse detection.

Quy tắc:
- Login → tạo family mới + session đầu tiên.
- Rotate → set revoked_at session cũ, tạo session mới cùng family, set replaced_by_id.
- Reuse detection: nếu submit token đã revoked → revoke TOÀN BỘ family.
"""
from __future__ import annotations

import uuid
from dataclasses import dataclass
from datetime import datetime, timedelta, timezone

from sqlalchemy import select, update
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.errors import unauthorized
from app.models.refresh_session import RefreshSession
from app.modules.auth.security import generate_refresh_token, hash_refresh_token


@dataclass
class RefreshResult:
    raw_token: str
    family_id: uuid.UUID
    expires_at: datetime


async def create_session_for_login(
    session: AsyncSession,
    *,
    user_id: str,
    raw_token: str,
    token_hash: str,
    ttl_seconds: int,
    user_agent: str | None = None,
    ip_address: str | None = None,
) -> uuid.UUID:
    """Tạo family mới + session đầu tiên. Return family_id."""
    family_id = uuid.uuid4()
    expires_at = datetime.now(timezone.utc) + timedelta(seconds=ttl_seconds)
    obj = RefreshSession(
        id=uuid.uuid4(),
        user_id=user_id,
        family_id=family_id,
        token_hash=token_hash,
        user_agent=user_agent,
        ip_address=ip_address,
        expires_at=expires_at,
    )
    session.add(obj)
    await session.flush()
    return family_id


async def rotate_session(
    session: AsyncSession,
    *,
    raw_token: str,
    family_id: uuid.UUID,
    ttl_seconds: int,
    user_agent: str | None = None,
    ip_address: str | None = None,
) -> RefreshResult:
    """Submit raw refresh → nếu match session active → rotate; nếu match revoked → reject.

    Raises:
      unauthorized: token không match / đã revoke / reuse detection.
    """
    submitted_hash = hash_refresh_token(raw_token)
    now = datetime.now(timezone.utc)

    # Tìm session match token_hash
    stmt = select(RefreshSession).where(
        RefreshSession.family_id == family_id,
        RefreshSession.token_hash == submitted_hash,
    )
    result = await session.execute(stmt)
    found = result.scalar_one_or_none()

    if found is None:
        raise unauthorized(
            detail="Refresh token không hợp lệ.",
            code="AUTH_REFRESH_INVALID",
        )

    if found.revoked_at is not None:
        # REUSE DETECTION → revoke toàn bộ family
        await session.execute(
            update(RefreshSession)
            .where(
                RefreshSession.family_id == family_id,
                RefreshSession.revoked_at.is_(None),
            )
            .values(revoked_at=now)
        )
        await session.flush()
        raise unauthorized(
            detail="Phát hiện reuse refresh token — toàn bộ session bị thu hồi.",
            code="AUTH_REFRESH_REUSE_DETECTED",
        )

    if found.expires_at <= now:
        raise unauthorized(
            detail="Refresh token đã hết hạn.",
            code="AUTH_REFRESH_EXPIRED",
        )

    # OK → rotate
    new_raw, new_hash = generate_refresh_token()
    new_session = RefreshSession(
        id=uuid.uuid4(),
        user_id=found.user_id,
        family_id=family_id,
        token_hash=new_hash,
        user_agent=user_agent,
        ip_address=ip_address,
        issued_at=now,
        expires_at=now + timedelta(seconds=ttl_seconds),
    )
    session.add(new_session)
    await session.flush()

    # Mark old as revoked + set replaced_by_id
    await session.execute(
        update(RefreshSession)
        .where(RefreshSession.id == found.id)
        .values(revoked_at=now, replaced_by_id=new_session.id)
    )
    await session.flush()

    return RefreshResult(
        raw_token=new_raw,
        family_id=family_id,
        expires_at=new_session.expires_at,
    )


async def revoke_family(
    session: AsyncSession,
    *,
    family_id: uuid.UUID,
) -> int:
    """Thu hồi toàn bộ family. Return số session bị revoke."""
    now = datetime.now(timezone.utc)
    stmt = (
        update(RefreshSession)
        .where(
            RefreshSession.family_id == family_id,
            RefreshSession.revoked_at.is_(None),
        )
        .values(revoked_at=now)
    )
    result = await session.execute(stmt)
    await session.flush()
    return result.rowcount or 0
```

```python
@router.post(
    "/refresh",
    response_model=None,
    status_code=status.HTTP_200_OK,
    summary="Rotate refresh token",
    responses={
        401: {"description": "Token invalid / expired / reuse detected"},
    },
)
async def refresh(
    request: Request,
    response: Response,
    session: AsyncSession = Depends(get_session),
) -> dict[str, Any]:
    """Submit refresh cookie → rotate (set new cookie)."""
    settings = get_settings()
    raw = request.cookies.get(settings.refresh_cookie_name)
    if not raw:
        raise unauthorized(
            detail="Thiếu refresh cookie.",
            code="AUTH_REFRESH_MISSING",
        )

    from app.modules.auth.refresh import rotate_session

    # Tìm family_id từ token_hash trước
    from app.models.refresh_session import RefreshSession
    from sqlalchemy import select
    from app.modules.auth.security import hash_refresh_token

    stmt = select(RefreshSession).where(
        RefreshSession.token_hash == hash_refresh_token(raw)
    )
    found = (await session.execute(stmt)).scalar_one_or_none()
    if found is None:
        raise unauthorized(
            detail="Refresh token không hợp lệ.",
            code="AUTH_REFRESH_INVALID",
        )

    user_agent = request.headers.get("user-agent")
    ip_address = request.client.host if request.client else None
    result = await rotate_session(
        session,
        raw_token=raw,
        family_id=found.family_id,
        ttl_seconds=settings.jwt_refresh_token_ttl_seconds,
        user_agent=user_agent,
        ip_address=ip_address,
    )
    await session.commit()

    # Reload user để trả về LoginResponse shape
    from app.models.user import User
    user = await session.get(User, found.user_id)
    if user is None:
        raise unauthorized(
            detail="User không tồn tại.",
            code="AUTH_USER_GONE",
        )
    user_public = UserPublic.model_validate(user)
    access_token = create_access_token(subject=user.id, role=user.role)

    response.set_cookie(
        key=settings.refresh_cookie_name,
        value=result.raw_token,
        max_age=settings.jwt_refresh_token_ttl_seconds,
        path=settings.refresh_cookie_path,
        secure=settings.refresh_cookie_secure,
        httponly=True,
        samesite="lax",
    )
    return _envelope(
        LoginResponse(
            access_token=access_token,
            expires_in=settings.jwt_access_token_ttl_seconds,
            user=user_public,
        ).model_dump()
    )

async def logout(
    request: Request,
    response: Response,
    session: AsyncSession = Depends(get_session),
) -> dict[str, Any]:
    """Submit refresh cookie → revoke family + clear cookie."""
    settings = get_settings()
    raw = request.cookies.get(settings.refresh_cookie_name)

    if raw:
        from app.modules.auth.refresh import revoke_family
        from app.models.refresh_session import RefreshSession
        from sqlalchemy import select
        from app.modules.auth.security import hash_refresh_token

        stmt = select(RefreshSession).where(
            RefreshSession.token_hash == hash_refresh_token(raw)
        )
        found = (await session.execute(stmt)).scalar_one_or_none()
        if found is not None:
            await revoke_family(session, family_id=found.family_id)
            await session.commit()

    response.delete_cookie(
        key=settings.refresh_cookie_name,
        path=settings.refresh_cookie_path,
        secure=settings.refresh_cookie_secure,
        httponly=True,
        samesite="lax",
    )
    return _envelope({})
```

```python
@model_validator(mode="after")
def _validate_production(self) -> "Settings":
    if self.app_env != "production":
        return self
    # Fail-closed checks
```
