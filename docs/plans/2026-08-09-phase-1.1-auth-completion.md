```python
"""Auth security primitives — argon2id (pwdlib) + JWT (PyJWT)."""
from __future__ import annotations

import hashlib
import secrets
import time
from typing import Any

import jwt
from pwdlib import PasswordHash

from app.core.config import get_settings

# argon2id recommended params (OWASP 2024).
_pwd = PasswordHash.recommended()

# Valid argon2id dummy hash for timing defense (email-not-found path).
# Hash of "dummy-password-never-used" — never used in practice.
DUMMY_PASSWORD_HASH: str = _pwd.hash("dummy-password-never-used")


def hash_password(plain: str) -> str:
    """Hash password với argon2id. Plain text KHÔNG log."""
    return _pwd.hash(plain)


def verify_password(plain: str, hashed: str) -> bool:
    """Verify plain password. Constant-time. False nếu hash malformed."""
    try:
        return _pwd.verify(plain, hashed)
    except (ValueError, TypeError):
        return False


# ---- Refresh token (opaque random, SHA-256 hash lưu DB) ----

def generate_refresh_token() -> tuple[str, str]:
    """Sinh opaque refresh token. Return (raw_token, sha256_hex_hash)."""
    raw = secrets.token_urlsafe(48)  # 384 bits entropy
    return raw, hashlib.sha256(raw.encode("utf-8")).hexdigest()


def hash_refresh_token(raw: str) -> str:
    """SHA-256 hex của raw refresh token."""
    return hashlib.sha256(raw.encode("utf-8")).hexdigest()


# ---- Access token (JWT HS256) ----

def create_access_token(*, subject: str, role: str, ttl_seconds: int | None = None) -> str:
    settings = get_settings()
    ttl = ttl_seconds if ttl_seconds is not None else settings.jwt_access_token_ttl_seconds
    now = int(time.time())
    payload: dict[str, Any] = {
        "sub": subject,
        "role": role,
        "iat": now,
        "exp": now + ttl,
        "iss": settings.app_name,
        "type": "access",
    }
    return jwt.encode(payload, settings.jwt_secret.get_secret_value(), algorithm=settings.jwt_algorithm)


def decode_access_token(token: str) -> dict[str, Any]:
    settings = get_settings()
    return jwt.decode(
        token,
        settings.jwt_secret.get_secret_value(),
        algorithms=[settings.jwt_algorithm],
        options={"require": ["exp", "sub", "role"]},
    )
```

## Commit 4: feat(auth): triển khai atomic refresh rotation + reuse detection + logout

**⚠️ 3 bug đã sửa trong design này:**
1. `rotate_session` nhận `raw_token` → lookup `family_id` từ DB (không client gửi).
2. Family revocation commit **trước** khi raise (không flush-then-raise).
3. Dùng `FOR UPDATE` hoặc atomic conditional update cho rotation.

### Step 3 — Create `app/modules/auth/refresh.py`

```python
"""Refresh token rotation + reuse detection (atomic, rollback-safe).

Key design principles:
1. family_id được lookup từ DB bằng token_hash, KHÔNG nhận từ client.
2. Family revocation commit TRƯỚC khi raise RotateError — không bị rollback.
3. Rotation dùng SELECT ... FOR UPDATE để chống concurrent refresh.
"""
from __future__ import annotations

import uuid
from dataclasses import dataclass
from datetime import datetime, timedelta, timezone

from sqlalchemy import select, update
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.errors import unauthorized
from app.models.refresh_session import RefreshSession
from app.modules.auth.security import generate_refresh_token, hash_refresh_token


class RotateError(Exception):
    """Refresh rotation failed — raised AFTER family revocation is committed."""

    def __init__(self, detail: str, code: str) -> None:
        super().__init__(detail)
        self.detail = detail
        self.code = code


@dataclass
class RefreshResult:
    raw_token: str
    family_id: uuid.UUID
    expires_at: datetime


async def create_session_for_login(
    session: AsyncSession,
    *,
    user_id: str,
    raw_token: str,
    token_hash: str,
    ttl_seconds: int,
    user_agent: str | None = None,
    ip_address: str | None = None,
) -> uuid.UUID:
    """Tạo family mới + session đầu tiên. Return family_id."""
    family_id = uuid.uuid4()
    expires_at = datetime.now(timezone.utc) + timedelta(seconds=ttl_seconds)
    obj = RefreshSession(
        id=uuid.uuid4(),
        user_id=user_id,
        family_id=family_id,
        token_hash=token_hash,
        user_agent=user_agent,
        ip_address=ip_address,
        expires_at=expires_at,
    )
    session.add(obj)
    await session.flush()
    return family_id


async def rotate_session(
    session: AsyncSession,
    *,
    raw_token: str,
    ttl_seconds: int,
    user_agent: str | None = None,
    ip_address: str | None = None,
) -> RefreshResult:
    """Atomic rotate: lookup → validate → revoke → create → commit.

    Raises RotateError (AFTER committing family revocation on reuse).
    """
    submitted_hash = hash_refresh_token(raw_token)
    now = datetime.now(timezone.utc)

    # Tìm session bằng token_hash (KHÔNG dùng family_id từ client)
    stmt = (
        select(RefreshSession)
        .where(RefreshSession.token_hash == submitted_hash)
        .with_for_update()
    )
    result = await session.execute(stmt)
    found = result.scalar_one_or_none()

    if found is None:
        raise RotateError(
            detail="Refresh token không hợp lệ.",
            code="AUTH_REFRESH_INVALID",
        )

    # 1. EXPIRED check
    if found.expires_at <= now:
        raise RotateError(
            detail="Refresh token đã hết hạn.",
            code="AUTH_REFRESH_EXPIRED",
        )

    # 2. REUSE DETECTION — revoked token
    if found.revoked_at is not None:
        # Commit family revocation TRƯỚC KHI raise
        await _revoke_family_no_flush(session, found.family_id, now)
        await session.commit()
        raise RotateError(
            detail="Phát hiện reuse refresh token — toàn bộ session bị thu hồi.",
            code="AUTH_REFRESH_REUSE_DETECTED",
        )

    # 3. OK — rotate: atomic UPDATE to mark old as revoked
    family_id = found.family_id
    new_raw, new_hash = generate_refresh_token()
    new_session = RefreshSession(
        id=uuid.uuid4(),
        user_id=found.user_id,
        family_id=family_id,
        token_hash=new_hash,
        user_agent=user_agent,
        ip_address=ip_address,
        issued_at=now,
        expires_at=now + timedelta(seconds=ttl_seconds),
    )
    session.add(new_session)

    # Atomic UPDATE: chỉ revoke row nếu chưa revoked (dùng FOR UPDATE lock)
    update_result = await session.execute(
        update(RefreshSession)
        .where(
            RefreshSession.id == found.id,
            RefreshSession.revoked_at.is_(None),  # double-check under lock
        )
        .values(revoked_at=now, replaced_by_id=new_session.id)
    )
    if update_result.rowcount == 0:
        # Concurrent request đã revoke row này → reuse detection
        await session.rollback()
        await _revoke_family_no_flush(session, family_id, now)
        await session.commit()
        raise RotateError(
            detail="Phát hiện reuse refresh token — toàn bộ session bị thu hồi.",
            code="AUTH_REFRESH_REUSE_DETECTED",
        )

    await session.commit()
    return RefreshResult(
        raw_token=new_raw,
        family_id=family_id,
        expires_at=new_session.expires_at,
    )


async def _revoke_family_no_flush(session: AsyncSession, family_id: uuid.UUID, now: datetime) -> None:
    """Revoke all active sessions in family. Does NOT flush — caller commits."""
    await session.execute(
        update(RefreshSession)
        .where(
            RefreshSession.family_id == family_id,
            RefreshSession.revoked_at.is_(None),
        )
        .values(revoked_at=now)
    )


async def revoke_family(
    session: AsyncSession,
    *,
    family_id: uuid.UUID,
) -> int:
    """Thu hồi toàn bộ family. Commit bên trong."""
    now = datetime.now(timezone.utc)
    await _revoke_family_no_flush(session, family_id, now)
    await session.commit()
    # Return rowcount — approximate, không hoàn hảo nhưng đủ audit
    return 0  # PostgreSQL trả rowcount nhưng asyncpg cần.execute.returning...


async def lookup_session_by_token(
    session: AsyncSession,
    raw_token: str,
) -> RefreshSession | None:
    """Lookup session bằng token_hash. Dùng cho /logout endpoint."""
    submitted_hash = hash_refresh_token(raw_token)
    stmt = select(RefreshSession).where(RefreshSession.token_hash == submitted_hash)
    result = await session.execute(stmt)
    return result.scalar_one_or_none()
```

```python
# app/modules/auth/router.py

from fastapi import APIRouter, Depends, Request, Response, status

@router.post("/login", response_model=None)
async def login(body: LoginRequest, response: Response, request: Request,
                session: AsyncSession = Depends(get_session)) -> dict[str, Any]:
    user = await authenticate(session, body.email, body.password, verify_password)
    if user is None:
        raise unauthorized(detail="Email hoặc mật khẩu không chính xác.",
                           code="AUTH_INVALID_CREDENTIALS")

    settings = get_settings()
    access_token = create_access_token(subject=user.id, role=user.role)
    user_public = UserPublic.model_validate(user)

    raw_refresh, refresh_hash = generate_refresh_token()
    await create_session_for_login(
        session,
        user_id=user.id,
        raw_token=raw_refresh,
        token_hash=refresh_hash,
        ttl_seconds=settings.jwt_refresh_token_ttl_seconds,
        user_agent=request.headers.get("user-agent"),
        ip_address=request.client.host if request.client else None,
    )
    await session.commit()

    response.set_cookie(
        key=settings.refresh_cookie_name, value=raw_refresh,
        max_age=settings.jwt_refresh_token_ttl_seconds,
        path=settings.refresh_cookie_path,
        secure=settings.refresh_cookie_secure,
        httponly=True, samesite="lax",
    )
    return _envelope(LoginResponse(
        access_token=access_token,
        expires_in=settings.jwt_access_token_ttl_seconds,
        user=user_public,
    ).model_dump())


@router.post("/refresh", response_model=None, status_code=status.HTTP_200_OK,
              responses={401: {"description": "Token invalid / expired / reuse detected"}})
async def refresh(request: Request, response: Response,
                  session: AsyncSession = Depends(get_session)) -> dict[str, Any]:
    settings = get_settings()
    raw = request.cookies.get(settings.refresh_cookie_name)
    if not raw:
        raise unauthorized(detail="Thiếu refresh cookie.", code="AUTH_REFRESH_MISSING")

    from app.modules.auth.refresh import rotate_session, lookup_session_by_token

    found = await lookup_session_by_token(session, raw)
    if found is None:
        raise unauthorized(detail="Refresh token không hợp lệ.", code="AUTH_REFRESH_INVALID")

    # Check inactive user
    if not found.user.is_active:
        from app.modules.auth.refresh import revoke_family
        await revoke_family(session, family_id=found.family_id)
        raise unauthorized(detail="Tài khoản đã bị vô hiệu hoá.",
                           code="AUTH_USER_INACTIVE")

    user_agent = request.headers.get("user-agent")
    ip = request.client.host if request.client else None
    try:
        result = await rotate_session(
            session, raw_token=raw, ttl_seconds=settings.jwt_refresh_token_ttl_seconds,
            user_agent=user_agent, ip_address=ip,
        )
    except RotateError as e:
        # Reuse/expiry đã được commit trong rotate_session
        raise unauthorized(detail=e.detail, code=e.code)

    access_token = create_access_token(subject=found.user_id, role=found.user.role)

    response.set_cookie(
        key=settings.refresh_cookie_name, value=result.raw_token,
        max_age=settings.jwt_refresh_token_ttl_seconds,
        path=settings.refresh_cookie_path,
        secure=settings.refresh_cookie_secure,
        httponly=True, samesite="lax",
    )
    return _envelope({
        "access_token": access_token,
        "expires_in": settings.jwt_access_token_ttl_seconds,
    })


@router.post("/logout", status_code=status.HTTP_204_NO_CONTENT)
async def logout(request: Request, response: Response,
                 session: AsyncSession = Depends(get_session)) -> None:
    settings = get_settings()
    raw = request.cookies.get(settings.refresh_cookie_name)

    if raw:
        from app.modules.auth.refresh import lookup_session_by_token, revoke_family
        found = await lookup_session_by_token(session, raw)
        if found is not None:
            await revoke_family(session, family_id=found.family_id)

    response.delete_cookie(
        key=settings.refresh_cookie_name,
        path=settings.refresh_cookie_path,
        secure=settings.refresh_cookie_secure,
        httponly=True, samesite="lax",
    )
```

**Key changes:**
- `/auth/refresh` trả `{access_token, expires_in}` (sync với OpenAPI).
- `/auth/logout` trả `204 No Content`.
- Logout idempotent (không token → vẫn clear cookie + 204).
- RotateError commit family revocation trước khi raise.
- User `is_active` check trước khi rotate.
