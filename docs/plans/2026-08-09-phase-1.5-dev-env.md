**Goal:** Hướng dẫn + verify script để dev mới (Windows 10) cài được Docker + Postgres + Redis + MinIO + oasdiff trong ~30-45 phút, sẵn sàng chạy `make up` cho Phase 2 BE.

- WSL2 (Ubuntu 22.04 LTS) + Docker Desktop for Windows với WSL2 backend.
- oasdiff binary (Go). Source: `https://github.com/oasdiff/oasdiff/releases/latest` (Windows AMD64 + Linux).
- Bash 4+ cho `verify-env.sh`, PowerShell 5.1+ cho `verify-env.ps1`.

**Out of scope (Phase 2+):**
- Code BE thật (`/documents` GET, RBAC filter).
- Celery worker.
- pgvector tables migration.
- oasdiff integrate vào CI.

## Task 2: Viết verify script (Bash + PowerShell)

**Step 1: Bash script `scripts/verify-env.sh`**

```bash
#!/usr/bin/env bash
# verify-env.sh — check Docker + Docker Compose + Postgres + Redis + MinIO + oasdiff.
# Chạy:  bash scripts/verify-env.sh
# Exit code 0 = tất cả pass, 1 = có ít nhất 1 fail.

set -u
PASS=0; FAIL=0
ok() { echo "  [OK]   $1"; PASS=$((PASS+1)); }
ko() { echo "  [FAIL] $1"; FAIL=$((FAIL+1)); }

echo "=== 1. Docker ==="
if command -v docker >/dev/null 2>&1; then
  ok "docker: $(docker --version)"
else
  ko "docker chưa cài — xem docs/setup/dev-environment.md §3"
fi

if command -v docker >/dev/null 2>&1 && docker compose version >/dev/null 2>&1; then
  ok "docker compose: $(docker compose version)"
else
  ko "docker compose chưa khả dụng"
fi

echo "=== 2. oasdiff ==="
if command -v oasdiff >/dev/null 2>&1; then
  ok "oasdiff: $(oasdiff --version 2>&1 | head -1)"
else
  ko "oasdiff chưa cài — xem docs/setup/dev-environment.md §5"
fi

echo "=== 3. Docker daemon ==="
if command -v docker >/dev/null 2>&1 && docker info >/dev/null 2>&1; then
  ok "Docker daemon đang chạy"
else
  ko "Docker daemon không phản hồi — mở Docker Desktop"
fi

echo "=== 4. Postgres reachability (optional, chỉ check khi stack đang up) ==="
if command -v docker >/dev/null 2>&1 && docker ps --format '{{.Names}}' 2>/dev/null | grep -q 'ctsv-postgres'; then
  if docker exec ctsv-postgres pg_isready -U ctsv_app >/dev/null 2>&1; then
    ok "Postgres ready (ctsv-postgres)"
  else
    ko "ctsv-postgres container chưa ready"
  fi
else
  echo "  [SKIP] Postgres container chưa chạy (chạy 'make up' trước)"
fi

echo "=== 5. Redis reachability ==="
if command -v docker >/dev/null 2>&1 && docker ps --format '{{.Names}}' 2>/dev/null | grep -q 'ctsv-redis'; then
  if docker exec ctsv-redis redis-cli ping >/dev/null 2>&1; then
    ok "Redis ready (ctsv-redis)"
  else
    ko "ctsv-redis container chưa ready"
  fi
else
  echo "  [SKIP] Redis container chưa chạy"
fi

echo "=== 6. MinIO reachability ==="
if command -v docker >/dev/null 2>&1 && docker ps --format '{{.Names}}' 2>/dev/null | grep -q 'ctsv-minio'; then
  if curl -fsSL http://localhost:9000/minio/health/live >/dev/null 2>&1; then
    ok "MinIO live (http://localhost:9000)"
  else
    ko "MinIO không phản hồi"
  fi
else
  echo "  [SKIP] MinIO container chưa chạy"
fi

echo
echo "=== TỔNG KẾT: $PASS pass, $FAIL fail ==="
[ "$FAIL" -eq 0 ]
```

**Step 2: PowerShell script `scripts/verify-env.ps1`**

```powershell
# verify-env.ps1 - check Docker + Docker Compose + Postgres + Redis + MinIO + oasdiff.
# Chạy:  powershell -ExecutionPolicy Bypass -File scripts/verify-env.ps1
# Exit code 0 = tất cả pass, 1 = có ít nhất 1 fail.

$pass = 0; $fail = 0
function Ok($msg) { Write-Host "  [OK]   $msg"; $script:pass++ }
function Ko($msg) { Write-Host "  [FAIL] $msg"; $script:fail++ }
function Skip($msg) { Write-Host "  [SKIP] $msg" }

Write-Host "=== 1. Docker ==="
try {
  $v = docker --version 2>$null
  if ($v) { Ok "docker: $v" } else { Ko "docker chưa cài" }
} catch { Ko "docker chưa cài - xem docs/setup/dev-environment.md §3" }

try {
  $v = docker compose version 2>$null
  if ($v) { Ok "docker compose: $v" } else { Ko "docker compose chưa khả dụng" }
} catch { Ko "docker compose chưa khả dụng" }

Write-Host "=== 2. oasdiff ==="
try {
  $v = oasdiff --version 2>$null
  if ($v) { Ok "oasdiff: $($v -split "`n" | Select-Object -First 1)" } else { Ko "oasdiff chưa cài" }
} catch { Ko "oasdiff chưa cài - xem docs/setup/dev-environment.md §5" }

Write-Host "=== 3. Docker daemon ==="
try {
  $info = docker info 2>$null
  if ($info) { Ok "Docker daemon đang chạy" } else { Ko "Docker daemon không phản hồi" }
} catch { Ko "Docker daemon không phản hồi - mở Docker Desktop" }

Write-Host "=== 4. Postgres reachability (chỉ check khi stack đang up) ==="
try {
  $pg = docker ps --format '{{.Names}}' 2>$null | Select-String -Pattern 'ctsv-postgres'
  if ($pg) {
    docker exec ctsv-postgres pg_isready -U ctsv_app 2>$null | Out-Null
    if ($LASTEXITCODE -eq 0) { Ok "Postgres ready (ctsv-postgres)" } else { Ko "ctsv-postgres không ready" }
  } else {
    Skip "Postgres container chưa chạy (chạy 'make up' trước)"
  }
} catch { Skip "Postgres container chưa chạy" }

Write-Host "=== 5. Redis reachability ==="
try {
  $r = docker ps --format '{{.Names}}' 2>$null | Select-String -Pattern 'ctsv-redis'
  if ($r) {
    docker exec ctsv-redis redis-cli ping 2>$null | Out-Null
    if ($LASTEXITCODE -eq 0) { Ok "Redis ready (ctsv-redis)" } else { Ko "ctsv-redis không ready" }
  } else {
    Skip "Redis container chưa chạy"
  }
} catch { Skip "Redis container chưa chạy" }

Write-Host "=== 6. MinIO reachability ==="
try {
  $m = docker ps --format '{{.Names}}' 2>$null | Select-String -Pattern 'ctsv-minio'
  if ($m) {
    $resp = Invoke-WebRequest -Uri 'http://localhost:9000/minio/health/live' -UseBasicParsing -TimeoutSec 3
    if ($resp.StatusCode -eq 200) { Ok "MinIO live (http://localhost:9000)" } else { Ko "MinIO trả $resp.StatusCode" }
  } else {
    Skip "MinIO container chưa chạy"
  }
} catch { Skip "MinIO container chưa chạy" }

Write-Host ""
Write-Host "=== TỔNG KẾT: $pass pass, $fail fail ==="
exit $fail
```
